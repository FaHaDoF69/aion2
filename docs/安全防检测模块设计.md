# 安全防检测模块设计

## 1. 总体安全策略

### 1.1 安全设计原则
- **零注入原则**: 不向游戏进程注入任何代码
- **API透明性**: 只使用标准Windows API
- **行为拟人化**: 模拟真实用户操作模式
- **可配置性**: 所有安全参数可动态调整
- **监控预警**: 实时监控系统状态和风险指标

### 1.2 防检测层次
```
┌─────────────────────────────────────────┐
│           应用层防检测                    │  ← 操作频率、业务逻辑
├─────────────────────────────────────────┤
│           行为层防检测                    │  ← 鼠标轨迹、操作节奏
├─────────────────────────────────────────┤
│           系统层防检测                    │  ← API调用、进程隐藏
├─────────────────────────────────────────┤
│           网络层防检测                    │  ← 流量特征、时序分析
└─────────────────────────────────────────┘
```

### 2.1 随机化引擎 (RandomizationEngine)

```csharp
public interface IRandomGenerator
{
    int Next(int min, int max);
    double NextGaussian(double mean, double stdDev);
    TimeSpan NextDelay(TimeSpan baseDelay, double variance = 0.3);
    Point NextOffset(Point center, int maxRadius);
    bool NextBool(double probability = 0.5);
}

public class CryptographicRandomGenerator : IRandomGenerator
{
    private readonly RandomNumberGenerator _rng;
    private readonly Random _fallbackRandom;
    
    public CryptographicRandomGenerator()
    {
        _rng = RandomNumberGenerator.Create();
        _fallbackRandom = new Random();
    }
    
    public double NextGaussian(double mean, double stdDev)
    {
        // Box-Muller变换生成正态分布随机数
        static double NextGaussianInternal()
        {
            double u1 = 1.0 - Random.Shared.NextDouble();
            double u2 = 1.0 - Random.Shared.NextDouble();
            return Math.Sqrt(-2.0 * Math.Log(u1)) * Math.Sin(2.0 * Math.PI * u2);
        }
        
        return mean + stdDev * NextGaussianInternal();
    }
    
    public TimeSpan NextDelay(TimeSpan baseDelay, double variance = 0.3)
    {
        var meanMs = baseDelay.TotalMilliseconds;
        var stdDev = meanMs * variance;
        var randomMs = Math.Max(100, NextGaussian(meanMs, stdDev)); // 最小100ms
        
        return TimeSpan.FromMilliseconds(randomMs);
    }
    
    public Point NextOffset(Point center, int maxRadius)
    {
        // 在圆形区域内生成随机偏移
        var angle = _fallbackRandom.NextDouble() * 2 * Math.PI;
        var radius = _fallbackRandom.NextDouble() * maxRadius;
        
        var offsetX = (int)(radius * Math.Cos(angle));
        var offsetY = (int)(radius * Math.Sin(angle));
        
        return new Point(center.X + offsetX, center.Y + offsetY);
    }
}
```

### 2.2 人性化鼠标控制器 (HumanLikeMouseController)

```csharp
public class HumanLikeMouseController : IMouseController
{
    private readonly IRandomGenerator _random;
    private readonly ILogger<HumanLikeMouseController> _logger;
    private Point _lastPosition;
    private DateTime _lastMoveTime;
    
    public async Task MoveToAsync(Point target, bool humanLike = true)
    {
        if (!humanLike)
        {
            SetCursorPos(target.X, target.Y);
            _lastPosition = target;
            return;
        }
        
        var path = GenerateHumanLikePath(_lastPosition, target);
        
        foreach (var step in path)
        {
            SetCursorPos(step.Point.X, step.Point.Y);
            
            // 根据移动距离调整延迟
            var delay = CalculateMoveDelay(step.Distance);
            await Task.Delay(delay);
        }
        
        _lastPosition = target;
        _lastMoveTime = DateTime.Now;
    }
    
    private List<MoveStep> GenerateHumanLikePath(Point start, Point end)
    {
        var steps = new List<MoveStep>();
        var distance = CalculateDistance(start, end);
        
        if (distance < 10)
        {
            // 短距离直接移动
            steps.Add(new MoveStep { Point = end, Distance = distance });
            return steps;
        }
        
        // 生成贝塞尔曲线路径
        var controlPoints = GenerateControlPoints(start, end);
        var pathPoints = BezierCurve.Generate(start, controlPoints[0], controlPoints[1], end, 
            Math.Max(10, (int)(distance / 20)));
        
        // 添加人性化特征
        for (int i = 0; i < pathPoints.Count; i++)
        {
            var point = pathPoints[i];
            
            // 添加轻微抖动
            if (_random.NextBool(0.1)) // 10%概率添加抖动
            {
                point = _random.NextOffset(point, 2);
            }
            
            // 计算移动距离
            var stepDistance = i > 0 ? CalculateDistance(pathPoints[i-1], point) : 0;
            
            steps.Add(new MoveStep { Point = point, Distance = stepDistance });
        }
        
        return steps;
    }
    
    private Point[] GenerateControlPoints(Point start, Point end)
    {
        var midX = (start.X + end.X) / 2;
        var midY = (start.Y + end.Y) / 2;
        var distance = CalculateDistance(start, end);
        
        // 控制点偏移量基于距离
        var maxOffset = Math.Min(100, distance / 4);
        
        var control1 = new Point(
            midX + _random.Next(-maxOffset, maxOffset),
            midY + _random.Next(-maxOffset, maxOffset)
        );
        
        var control2 = new Point(
            midX + _random.Next(-maxOffset, maxOffset),
            midY + _random.Next(-maxOffset, maxOffset)
        );
        
        return new[] { control1, control2 };
    }
    
    public async Task ClickAsync(Point position, MouseButton button = MouseButton.Left)
    {
        // 1. 移动到目标位置
        await MoveToAsync(position, true);
        
        // 2. 点击前的微小暂停（模拟瞄准时间）
        var aimDelay = _random.Next(50, 200);
        await Task.Delay(aimDelay);
        
        // 3. 执行点击
        var clickDuration = _random.Next(80, 180);
        MouseDown(button);
        await Task.Delay(clickDuration);
        MouseUp(button);
        
        // 4. 点击后的微小移动（模拟手部自然移动）
        var afterClickOffset = _random.NextOffset(position, 3);
        await Task.Delay(_random.Next(50, 150));
        await MoveToAsync(afterClickOffset, false);
        
        _logger.LogDebug("执行点击: {Position}, 持续时间: {Duration}ms", position, clickDuration);
    }
}

public class MoveStep
{
    public Point Point { get; set; }
    public double Distance { get; set; }
}
```

```csharp
public class HumanLikeKeyboardController : IKeyboardController
{
    private readonly IRandomGenerator _random;
    
    public async Task TypeTextAsync(string text, bool humanLike = true)
    {
        if (!humanLike)
        {
            foreach (char c in text)
            {
                SendChar(c);
            }
            return;
        }
        
        // 人性化打字
        for (int i = 0; i < text.Length; i++)
        {
            char c = text[i];
            
            // 模拟打字错误和修正
            if (_random.NextBool(0.02)) // 2%概率打错
            {
                await SimulateTypingError(c);
            }
            
            // 发送字符
            SendChar(c);
            
            // 计算打字间隔
            var typingDelay = CalculateTypingDelay(c, i < text.Length - 1 ? text[i + 1] : '\0');
            await Task.Delay(typingDelay);
        }
    }
    
    private async Task SimulateTypingError(char correctChar)
    {
        // 打一个错误字符
        var wrongChar = GetNearbyKey(correctChar);
        SendChar(wrongChar);
        
        // 暂停（发现错误）
        await Task.Delay(_random.Next(200, 800));
        
        // 删除错误字符
        KeyPress(Keys.Back);
        await Task.Delay(_random.Next(100, 300));
    }
    
    private int CalculateTypingDelay(char current, char next)
    {
        // 基础打字速度：每分钟200-300字符
        var baseDelay = _random.Next(200, 300);
        
        // 根据字符类型调整
        if (char.IsUpper(current)) baseDelay += 50; // 大写字母需要Shift
        if (char.IsDigit(current)) baseDelay += 20; // 数字稍慢
        if (char.IsPunctuation(current)) baseDelay += 30; // 标点符号
        
        // 连续相同字符稍快
        if (current == next) baseDelay = (int)(baseDelay * 0.8);
        
        // 添加随机变化
        return (int)_random.NextGaussian(baseDelay, baseDelay * 0.2);
    }
}
```

### 3.1 频率限制器 (FrequencyLimiter)

```csharp
public interface IFrequencyLimiter
{
    Task<bool> CanPerformOperation(string operationType);
    Task RecordOperation(string operationType);
    Task<TimeSpan> GetNextAllowedTime(string operationType);
    void UpdateLimits(Dictionary<string, FrequencyLimit> newLimits);
}

public class AdaptiveFrequencyLimiter : IFrequencyLimiter
{
    private readonly Dictionary<string, Queue<DateTime>> _operationHistory;
    private readonly Dictionary<string, FrequencyLimit> _limits;
    private readonly IRandomGenerator _random;
    
    public async Task<bool> CanPerformOperation(string operationType)
    {
        if (!_limits.ContainsKey(operationType))
            return true;
        
        var limit = _limits[operationType];
        var history = GetOperationHistory(operationType);
        
        // 清理过期记录
        CleanupExpiredRecords(history, limit.TimeWindow);
        
        // 检查频率限制
        if (history.Count >= limit.MaxOperations)
        {
            return false;
        }
        
        // 检查最小间隔
        if (history.Any())
        {
            var lastOperation = history.Last();
            var timeSinceLastOperation = DateTime.Now - lastOperation;
            
            if (timeSinceLastOperation < limit.MinInterval)
            {
                return false;
            }
        }
        
        return true;
    }
    
    public async Task RecordOperation(string operationType)
    {
        var history = GetOperationHistory(operationType);
        history.Enqueue(DateTime.Now);
        
        // 记录到日志用于分析
        _logger.LogDebug("记录操作: {OperationType}, 当前频率: {Count}/{Window}", 
            operationType, history.Count, _limits[operationType].TimeWindow);
    }
    
    private Queue<DateTime> GetOperationHistory(string operationType)
    {
        if (!_operationHistory.ContainsKey(operationType))
        {
            _operationHistory[operationType] = new Queue<DateTime>();
        }
        return _operationHistory[operationType];
    }
}

public class FrequencyLimit
{
    public int MaxOperations { get; set; }
    public TimeSpan TimeWindow { get; set; }
    public TimeSpan MinInterval { get; set; }
    public bool AdaptiveEnabled { get; set; }
}
```

```csharp
{
    private readonly Dictionary<string, BehaviorPattern> _patterns;

    public BehaviorPatternManager()

    }
    
    private Dictionary<string, BehaviorPattern> InitializePatterns()
    {
        return new Dictionary<string, BehaviorPattern>
        {
            ["Conservative"] = new BehaviorPattern
            {
                Name = "Conservative",
                OperationInterval = TimeSpan.FromSeconds(8),
                OperationVariance = 0.4,
                MaxOperationsPerHour = 30,
                BreakProbability = 0.15,
                BreakDuration = TimeSpan.FromMinutes(3)
            },
            ["Normal"] = new BehaviorPattern
            {
                Name = "Normal",
                OperationInterval = TimeSpan.FromSeconds(5),
                OperationVariance = 0.3,
                MaxOperationsPerHour = 50,
                BreakProbability = 0.1,
                BreakDuration = TimeSpan.FromMinutes(2)
            },
            ["Aggressive"] = new BehaviorPattern
            {
                Name = "Aggressive",
                OperationInterval = TimeSpan.FromSeconds(3),
                OperationVariance = 0.2,
                MaxOperationsPerHour = 80,
                BreakProbability = 0.05,
                BreakDuration = TimeSpan.FromMinutes(1)
            }
        };
    }
    
    public async Task<bool> ShouldTakeBreak()
    {
        var sessionDuration = DateTime.Now - _sessionStartTime;
        var timeSinceLastBreak = DateTime.Now - _lastBreakTime;
        
        // 基于时间的强制休息
        if (sessionDuration.TotalHours >= 2 && timeSinceLastBreak.TotalMinutes >= 30)
        {
            return true;
        }
        
        // 随机休息
        if (timeSinceLastBreak.TotalMinutes >= 10)
        {
            return _random.NextBool(_currentPattern.BreakProbability);
        }
        
        return false;
    }
    
    public async Task SimulateUserBreak()
    {
        var breakDuration = _random.NextDelay(_currentPattern.BreakDuration, 0.5);
        
        _logger.LogInformation("开始模拟用户休息，持续时间: {Duration}", breakDuration);
        
        // 在休息期间可以执行一些"人性化"操作
        await SimulateBreakActivities(breakDuration);
        
        _lastBreakTime = DateTime.Now;
        
        _logger.LogInformation("用户休息结束");
    }
    
    private async Task SimulateBreakActivities(TimeSpan breakDuration)
    {
        var endTime = DateTime.Now + breakDuration;
        
        while (DateTime.Now < endTime)
        {
            // 随机执行一些"无关"操作
            if (_random.NextBool(0.3))
            {
                // 模拟查看其他窗口
                await SimulateWindowSwitch();
            }
            
            await Task.Delay(_random.Next(5000, 15000));
        }
    }
    
    public void UpdatePatternBasedOnTime()
    {
        var hour = DateTime.Now.Hour;
        
        // 根据时间调整行为模式
        string newPatternName = hour switch
        {
            >= 9 and <= 11 => "Aggressive",   // 上午活跃时间
            >= 14 and <= 16 => "Normal",      // 下午正常时间
            >= 20 and <= 22 => "Conservative", // 晚上保守时间
            _ => "Conservative"                // 其他时间保守
        };
        
        if (_currentPattern.Name != newPatternName)
        {
            _currentPattern = _patterns[newPatternName];
            _logger.LogInformation("切换行为模式: {PatternName}", newPatternName);
        }
    }
}

    public double OperationVariance { get; set; }
    public int MaxOperationsPerHour { get; set; }
    public double BreakProbability { get; set; }
```

### 4.1 风险监控器 (RiskMonitor)

```csharp
public interface IRiskMonitor
{
    Task<RiskLevel> AssessCurrentRisk();
    Task<List<RiskAlert>> GetActiveAlerts();
    Task MonitorSystemHealth();
    event EventHandler<RiskAlert> RiskAlertRaised;
}

public class ComprehensiveRiskMonitor : IRiskMonitor
{
    private readonly List<IRiskDetector> _detectors;
    private readonly ILogger<ComprehensiveRiskMonitor> _logger;
    
    public event EventHandler<RiskAlert> RiskAlertRaised;
    
    public async Task<RiskLevel> AssessCurrentRisk()
    {
        var risks = new List<RiskAssessment>();
        
        // 执行所有风险检测器
        foreach (var detector in _detectors)
        {
            try
            {
                var assessment = await detector.AssessRisk();
                risks.Add(assessment);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "风险检测器执行失败: {DetectorType}", detector.GetType().Name);
            }
        }
        
        // 计算综合风险等级
        return CalculateOverallRisk(risks);
    }
    
    private RiskLevel CalculateOverallRisk(List<RiskAssessment> assessments)
    {
        if (!assessments.Any()) return RiskLevel.Unknown;
        
        var highRiskCount = assessments.Count(a => a.Level == RiskLevel.High);
        var mediumRiskCount = assessments.Count(a => a.Level == RiskLevel.Medium);
        
        if (highRiskCount > 0) return RiskLevel.High;
        if (mediumRiskCount >= 2) return RiskLevel.High;
        if (mediumRiskCount > 0) return RiskLevel.Medium;
        
        return RiskLevel.Low;
    }
}
```
